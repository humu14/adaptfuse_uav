% Bibliography formatting fixes for biblatex-ieee:
% (1) put a period between doi, arXiv id and URL ("doi: X. arXiv: Y.").
%     ieee's DOI format hides the DOI from the punctuation tracker, so a
%     literal period is used after it.
\newcommand*{\bibdoisep}{\iffieldundef{doi}{\addperiod\space}{.\space}}
\renewbibmacro*{doi+eprint+url}{%
  \setunit{\addperiod\space}%
  \iftoggle{bbx:doi}{\printfield{doi}}{}%
  \setunit{\bibdoisep}%
  \iftoggle{bbx:eprint}{\usebibmacro{eprint}}{}%
  \setunit{\bibdoisep}%
  \iftoggle{bbx:url}{\usebibmacro{url+urldate}}{}}
% (2) a title ending in "?" must not capitalise the following "in"
\DeclareCapitalPunctuation{.!}
